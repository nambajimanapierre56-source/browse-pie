function go() { let input = document.getElementById('urlInput').value.trim(); let url = ""; // Reba niba ibyo yanditse ari link if (input.startsWith('http://') || input.startsWith('https://')) { url = input; } else if (input.includes('.') && !input.includes(' ')) { url = 'https://' + input; } else { // Koresha Wiby cyangwa Wikipedia kuko zo zemera iframe neza kurusha Bing url = 'https://wiby.me' + encodeURIComponent(input); } document.getElementById('viewport').src = url; }
